{{- /* Newer/older links computed from date order, so they behave the same on every Hugo version. */ -}} {{- $cur := . }} {{- $pages := (where site.RegularPages "Section" "posts").ByDate.Reverse }} {{- $idx := -1 }} {{- range $i, $p := $pages }}{{ if eq $p.RelPermalink $cur.RelPermalink }}{{ $idx = $i }}{{ end }}{{ end }} {{- if ge $idx 0 }} {{- $newer := false }} {{- $older := false }} {{- if gt $idx 0 }}{{ $newer = index $pages (sub $idx 1) }}{{ end }} {{- if lt $idx (sub (len $pages) 1) }}{{ $older = index $pages (add $idx 1) }}{{ end }} {{- if or $newer $older }}
{{- with $older }} « older {{ .Title }} {{- end }} {{- with $newer }} newer » {{ .Title }} {{- end }}
{{- end }} {{- end }}
